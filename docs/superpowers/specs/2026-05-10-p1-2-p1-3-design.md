#### `indicators.py` — 技术指标计算

```python
import pandas as pd
import numpy as np

def calculate_rsi(closes: pd.Series, period: int = 14) -> pd.Series:
    """RSI = 100 - 100/(1+RS)，默认14日"""
    delta = closes.diff()
    gain = delta.where(delta > 0, 0)
    loss = -delta.where(delta < 0, 0)
    avg_gain = gain.rolling(period).mean()
    avg_loss = loss.rolling(period).mean()
    rs = avg_gain / avg_loss
    return 100 - (100 / (1 + rs))

def calculate_macd(closes: pd.Series, fast: int = 12, slow: int = 26, signal: int = 9):
    """返回 (macd_line, signal_line, histogram) 三元组"""
    ema_fast = closes.ewm(span=fast).mean()
    ema_slow = closes.ewm(span=slow).mean()
    macd_line = ema_fast - ema_slow
    signal_line = macd_line.ewm(span=signal).mean()
    histogram = macd_line - signal_line
    return macd_line, signal_line, histogram

def calculate_kdj(highs: pd.Series, lows: pd.Series, closes: pd.Series,
                   n: int = 9, m1: int = 3, m2: int = 3):
    """KDJ(9,3,3) — 随机指标"""
    lowest_low = lows.rolling(n).min()
    highest_high = highs.rolling(n).max()
    rsv = (closes - lowest_low) / (highest_high - lowest_low) * 100
    k = rsv.ewm(com=m1-1).mean()
    d = k.ewm(com=m2-1).mean()
    j = 3 * k - 2 * d
    return k, d, j

def calculate_wr(highs: pd.Series, lows: pd.Series, closes: pd.Series, period: int = 14):
    """WR = (HHV-HHV)/HHV*100，背离指标"""
    highest_high = highs.rolling(period).max()
    return (highest_high - closes) / (highest_high - lows) * 100

def calculate_obv(closes: pd.Series, volumes: pd.Series) -> pd.Series:
    """OBV = 累计成交量（价格上升时加，下降时减）"""
    direction = np.sign(closes.diff())
    return (direction * volumes).cumsum()

def calculate_bollinger_bands(closes: pd.Series, period: int = 20, num_std: float = 2):
    """布林带 = MA ± num_std * STD"""
    mid = closes.rolling(period).mean()
    std = closes.rolling(period).std()
    upper = mid + num_std * std
    lower = mid - num_std * std
    return upper, mid, lower
```

### 2.2 搜索结果缓存（`src/search_pkg/manager.py`）

**TTL：** 5 分钟（`SEARCH_CACHE_TTL = 300`）

```python
import time
from typing import Optional, Dict, Tuple

class SearchCache:
    """搜索结果缓存，5 分钟 TTL"""
    _cache: Dict[str, Tuple[str, float]] = {}  # {query: (result_json, timestamp)}

    def get(self, query: str) -> Optional[str]:
        if query in self._cache:
            result, ts = self._cache[query]
            if time.time() - ts < SEARCH_CACHE_TTL:
                return result
            del self._cache[query]
        return None

    def set(self, query: str, result: str):
        self._cache[query] = (result, time.time())

    def clear_expired(self):
        now = time.time()
        expired = [q for q, (_, ts) in self._cache.items() if now - ts >= SEARCH_CACHE_TTL]
        for q in expired:
            del self._cache[q]

SEARCH_CACHE_TTL = 300  # 5 分钟
```

### 2.3 Per-Fetcher 速率限制追踪（`src/data_provider/base.py`）

```python
import time
from typing import Dict

class RateLimiter:
    """每个 fetcher 的速率追踪，支持动态调整"""
    _last_request_time: Dict[str, float] = {}
    _intervals: Dict[str, float] = {}  # fetcher_name -> current min_interval

    def __init__(self, fetcher_name: str, base_interval: float = 1.0):
        self._fetcher_name = fetcher_name
        self._base_interval = base_interval
        self._intervals.setdefault(fetcher_name, base_interval)

    def wait_if_needed(self):
        elapsed = time.time() - self._last_request_time.get(self._fetcher_name, 0)
        interval = self._intervals.get(self._fetcher_name, self._base_interval)
        if elapsed < interval:
            time.sleep(interval - elapsed)
        self._last_request_time[self._fetcher_name] = time.time()

    def record_rate_limit_hit(self, retry_after: float = 1.0):
        """被限流时动态降低频率（增加间隔）"""
        current = self._intervals.get(self._fetcher_name, self._base_interval)
        self._intervals[self._fetcher_name] = min(current * 1.5, retry_after, 30.0)
        logger.warning(f"[{self._fetcher_name}] 限流，增加间隔到 {self._intervals[self._fetcher_name]:.1f}s")

    def decay(self):
        """每 10 分钟衰减回正常值（乘以 0.95）"""
        current = self._intervals.get(self._fetcher_name, self._base_interval)
        self._intervals[self._fetcher_name] = max(current * 0.95, self._base_interval)
```

**使用：** 每个 fetcher 的 `fetch()` 方法第一行调用 `self._rate_limiter.wait_if_needed()`
**衰减：** main.py 启动时后台线程每 10 分钟调用 `RateLimiter.decay()` 一次

### 2.4 AI API 429 处理（`src/analyzer.py`）

```python
from tenacity import retry, stop_after_attempt, wait_exponential, retry_if_exception
import httpx

def _is_rate_limit(e) -> bool:
    return isinstance(e, httpx.HTTPStatusError) and e.response.status_code == 429

class GeminiAnalyzer:
    def __init__(self, config):
        self._models = config.get("models", ["gemini-2.0-flash", "gemini-1.5-flash"])
        self._model_index = 0

    @property
    def current_model(self) -> str:
        return self._models[self._model_index]

    @tenacity.retry(
        retry=retry_if_exception(_is_rate_limit),
        wait=wait_exponential(multiplier=2, min=4, max=60),
        stop=stop_after_attempt(5),
        before_sleep=lambda retry_state: logger.warning(
            f"429 rate limit hit, retry #{retry_state.attempt_number}, "
            f"waiting {retry_state.next_action.sleep}s"
        )
    )
    async def analyze(self, code, context):
        response = await self._call_api(self.current_model, code, context)
        return response
```

**Fallback model 切换：** `src/analyzer.py` 新增 `switch_to_next_model()` 方法，429 时将 `self._model_index` 向后移动，环回到列表开头。

### 2.5 网络降级策略

**数据源优先级：** AkshareFetcher > YFinanceFetcher > EfinanceFetcher

**降级流程：**
```
try AkshareFetcher.fetch()
except NetworkError:
    try YFinanceFetcher.fetch()
    except NetworkError:
        try EfinanceFetcher.fetch()
        except NetworkError:
            # 所有数据源都失败
            if has_recent_cache():
                return cached_data_with_stale_warning()
            else:
                raise DataSourceError("所有数据源不可用")
```

**缓存数据判定（`has_recent_cache`）：**
- 缓存数据时间 <= 1 个交易日 → 可用（显示"数据可能过时"）
- 缓存数据时间 > 1 个交易日 → 强制提示刷新失败

**TUI/GUI 展示：**
- 数据过时：状态栏显示 `⚠️ 数据可能过时（缓存于 XX:XX）`
- 所有源失败：弹窗提示，不阻塞其他操作
